#pragma once
#include <unordered_map>
#include <cstdint>
#include <vector>
#include "node.h"

namespace entities {
    class Graph {
    private:
        std::unordered_map<uint64_t, Node> nodes;
        std::vector<std::pair<uint64_t, uint64_t>> edgeCreationOrder;
        std::vector<uint64_t> nodeCreationOrder;

    public:
        Graph() = default;

        const std::unordered_map<uint64_t, Node>& getNodes() const { return nodes; }
        const std::vector<std::pair<uint64_t, uint64_t>>& getEdgeCreationOrder() const { return edgeCreationOrder; }
        const std::vector<uint64_t>& getNodeCreationOrder() const { return nodeCreationOrder; }

        uint64_t addNode(float x, float y) {
            Node newNode(x, y);
            uint64_t id = newNode.getID();
            auto [it, inserted] = nodes.try_emplace(id, std::move(newNode));
            if (inserted) nodeCreationOrder.push_back(id);
            return id;
        }

        void addEdge(uint64_t id_A, uint64_t id_B, float weight, int angle = 0) {
            if (id_A == id_B) return;
            auto itA = nodes.find(id_A);
            auto itB = nodes.find(id_B);
            if (itA == nodes.end() || itB == nodes.end()) return;

            if (itA->second.getNeighbors().count(id_B)) return;   // O(1)

            itA->second.addNeighbor(id_B, weight);
            itB->second.addNeighbor(id_A, weight);

            if (id_A < id_B)
                edgeCreationOrder.emplace_back(id_A, id_B);
            else
                edgeCreationOrder.emplace_back(id_B, id_A);
        }

        bool hasEdge(uint64_t id_A, uint64_t id_B) const {
            auto itA = nodes.find(id_A);
            if (itA != nodes.end()) {
                return itA->second.getNeighbors().count(id_B) > 0;
            }
            return false;
        }

        void deleteEdge(uint64_t id_A, uint64_t id_B) {
            auto itA = nodes.find(id_A);
            auto itB = nodes.find(id_B);
            if (itA != nodes.end() && itB != nodes.end()) {
                itA->second.removeEdge(id_B);
                itB->second.removeEdge(id_A);
            }
        }

        void deleteNode(uint64_t targetID) {
            auto targetIt = nodes.find(targetID);
            if (targetIt == nodes.end()) return;
            for (uint64_t neighborID : targetIt->second.getNeighbors()) {
                auto neighborIt = nodes.find(neighborID);
                if (neighborIt != nodes.end()) {
                    neighborIt->second.removeEdge(targetID);
                }
            }
            nodes.erase(targetIt);
        }
    };
}

        float getX() const { return x; }
        float getY() const { return y; }

        const std::unordered_set<uint64_t>& getNeighbors() const { return neighbors; }
        float getWeightTo(uint64_t neighbor) const {
            auto it = weights.find(neighbor);
            return (it != weights.end()) ? it->second : 0.0f;

            neighbors.insert(neighbor);
            weights[neighbor] = weight;
        }

        void removeEdge(uint64_t targetDestinationID);
    };

// entities/algorithms/ChaosBlend.hpp
#pragma once
#include <SFML/Graphics/Rect.hpp>
#include <functional>
#include <vector>
#include <random>
#include "SeedManager.hpp"

namespace entities::algorithms::loom {

    // Firma comÃºn de todos tus algoritmos de dibujo
    using GeneratorFn = std::function<void(Graph&, sf::FloatRect /*bounds*/)>;

    // â”€â”€â”€ Nodo BSP â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
    struct BSPNode {
        sf::FloatRect region;
        std::unique_ptr<BSPNode> left, right;
        int generatorIndex{ -1 }; // -1 = nodo interior

        bool isLeaf() const { return !left && !right; }
    };

    // â”€â”€â”€ Ãrbol BSP â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
    /**
     * @brief Divide el canvas recursivamente hasta maxDepth y asigna
     *        un generador fractal a cada hoja.
     *
     * @param maxDepth   Profundidad mÃ¡xima de subdivisiÃ³n (recomendado 3-4)
     * @param minSize    TamaÃ±o mÃ­nimo de celda en pÃ­xeles (evita celdas triviales)
     * @param numGenerators  Cantidad de algoritmos disponibles
     */
    std::unique_ptr<BSPNode> buildBSP(
        const sf::FloatRect& bounds,
        int maxDepth,
        float minSize,
        int numGenerators,
        std::mt19937& rng)
    {
        auto node = std::make_unique<BSPNode>();
        node->region = bounds;

        // CondiciÃ³n de parada
        if (maxDepth == 0 ||
            bounds.width < minSize * 2 ||
            bounds.height < minSize * 2)
        {
            std::uniform_int_distribution<int> pick(0, numGenerators - 1);
            node->generatorIndex = pick(rng);
            return node;
        }

        // Decidir si corte horizontal o vertical
        std::bernoulli_distribution coin(0.5);
        bool splitH = coin(rng); // true = dividir horizontalmente

        // Ratio de corte con algo de variaciÃ³n para evitar grillas perfectas
        std::uniform_real_distribution<float> ratio(0.35f, 0.65f);
        float r = ratio(rng);

        sf::FloatRect leftRect = bounds, rightRect = bounds;
        if (splitH) {
            float cutY = bounds.top + bounds.height * r;
            leftRect = { bounds.left, bounds.top,   bounds.width, cutY - bounds.top };
            rightRect = { bounds.left, cutY,          bounds.width, bounds.top + bounds.height - cutY };
        }
        else {
            float cutX = bounds.left + bounds.width * r;
            leftRect = { bounds.left, bounds.top,  cutX - bounds.left, bounds.height };
            rightRect = { cutX,        bounds.top,  bounds.left + bounds.width - cutX, bounds.height };
        }

        node->left = buildBSP(leftRect, maxDepth - 1, minSize, numGenerators, rng);
        node->right = buildBSP(rightRect, maxDepth - 1, minSize, numGenerators, rng);
        return node;
    }

    // â”€â”€â”€ Recorrido del Ã¡rbol: aplica generadores a las hojas â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
    void applyGenerators(
        const BSPNode* node,
        Graph& g,
        const std::vector<GeneratorFn>& generators)
    {
        if (!node) return;
        if (node->isLeaf()) {
            generators[node->generatorIndex](g, node->region);
            return;
        }
        applyGenerators(node->left.get(), g, generators);
        applyGenerators(node->right.get(), g, generators);
    }

    // â”€â”€â”€ Punto de entrada del modo ChaosBlend â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
    /**
     * @brief Mode 10 â€” Chaos Blend
     *
     * Divide el canvas en subregiones usando BSP y renderiza un fractal
     * diferente en cada celda. Toda la geometrÃ­a va al mismo grafo g.
     *
     * @param generators  Todos tus algoritmos existentes empaquetados como lambdas
     * @param bounds      Ãrea total de dibujo (LOOM_WIDTH x LOOM_HEIGHT - MARGIN)
     */
    void chaosBlend(
        Graph& g,
        const std::vector<GeneratorFn>& generators,
        const sf::FloatRect& bounds)
    {
        auto& rng = SeedManager::instance().rng();

        // Selecciona entre 2 y min(4, total) generadores para esta ejecuciÃ³n
        int maxGen = static_cast<int>(generators.size());
        std::uniform_int_distribution<int> nDist(2, std::min(4, maxGen));
        int chosenN = nDist(rng);

        // Subconjunto aleatorio de generadores
        std::vector<int> indices(maxGen);
        std::iota(indices.begin(), indices.end(), 0);
        std::shuffle(indices.begin(), indices.end(), rng);
        indices.resize(chosenN);

        std::vector<GeneratorFn> selected;
        selected.reserve(chosenN);
        for (int i : indices) selected.push_back(generators[i]);

        // Construye el BSP con profundidad basada en el nÃºmero elegido
        int depth = (chosenN <= 2) ? 2 : 3;
        auto tree = buildBSP(bounds, depth, 80.f, chosenN, rng);

        // Aplica cada generador a su regiÃ³n
        applyGenerators(tree.get(), g, selected);
    }

} // namespace entities::algorithms::loom

    void expandSierpinski(Graph& g, int depth, int outerLayers,
        float x1, float y1, float x2, float y2, float x3, float y3) {
        sierpinski(g, depth, x1, y1, x2, y2, x3, y3);

        float ax = x1, ay = y1, bx = x2, by = y2, cx = x3, cy = y3;
        for (int layer = 1; layer <= outerLayers; ++layer) {
            float axNew = bx + cx - ax;
            float ayNew = by + cy - ay;
            float bxNew = cx + ax - bx;
            float byNew = cy + ay - by;
            float cxNew = ax + bx - cx;
            float cyNew = ay + by - cy;

            sierpinski(g, depth, axNew, ayNew, bx, by, cx, cy);
            sierpinski(g, depth, ax, ay, bxNew, byNew, cx, cy);
            sierpinski(g, depth, ax, ay, bx, by, cxNew, cyNew);

            ax = axNew; ay = ayNew;
            bx = bxNew; by = byNew;
            cx = cxNew; cy = cyNew;
        }
    }

    void expandAtVertices(Graph& g, int depth,
        float x1, float y1, float x2, float y2, float x3, float y3,
        int outerDepth) {
        if (outerDepth < 0) outerDepth = depth;

        sierpinski(g, depth, x1, y1, x2, y2, x3, y3);

        struct Tri {
            float ax, ay, bx, by, cx, cy;
        };
        Tri tA = { x1, y1, 2 * x1 - x2, 2 * y1 - y2, 2 * x1 - x3, 2 * y1 - y3 };
        Tri tB = { x2, y2, 2 * x2 - x3, 2 * y2 - y3, 2 * x2 - x1, 2 * y2 - y1 };
        Tri tC = { x3, y3, 2 * x3 - x1, 2 * y3 - y1, 2 * x3 - x2, 2 * y3 - y2 };

        std::mutex mergeMutex;

        auto threadWork = [&g, &mergeMutex, outerDepth](const Tri& tri) {
            Graph localGraph;
            sierpinski(localGraph, outerDepth, tri.ax, tri.ay, tri.bx, tri.by, tri.cx, tri.cy);

            std::lock_guard<std::mutex> lock(mergeMutex);
            // Insertar nodos locales en el grafo global
            for (const auto& [id, node] : localGraph.getNodes()) {
                uint64_t newId = g.addNode(node.getX(), node.getY());
                // Insertar aristas usando los vecinos del nodo local
                for (uint64_t neighborId : node.getNeighbors()) {
                    auto itNeighbor = localGraph.getNodes().find(neighborId);
                    if (itNeighbor != localGraph.getNodes().end()) {
                        uint64_t globalNeighborId = g.addNode(itNeighbor->second.getX(), itNeighbor->second.getY());
                        float weight = node.getWeightTo(neighborId);
                        g.addEdge(newId, globalNeighborId, weight);
                    }
                }
            }
            };

        std::thread thA(threadWork, tA);
        std::thread thB(threadWork, tB);
        std::thread thC(threadWork, tC);

        thA.join();
        thB.join();
        thC.join();
    }

    int computeOutCode(float x, float y, const sf::FloatRect& rect) {
        int code = INSIDE;
        if (x < rect.position.x)      code |= LEFT;
        else if (x > rect.position.x + rect.size.x) code |= RIGHT;
        if (y < rect.position.y)      code |= TOP;
        else if (y > rect.position.y + rect.size.y) code |= BOTTOM;
        return code;
    }

    std::optional<std::pair<std::pair<float, float>, std::pair<float, float>>>
        clipSegment(float x1, float y1, float x2, float y2, const sf::FloatRect& rect,
            std::vector<std::pair<float, float>>& intersections) {
        float minX = rect.position.x;
        float maxX = rect.position.x + rect.size.x;
        float minY = rect.position.y;
        float maxY = rect.position.y + rect.size.y;

        sf::FloatRect tempRect(sf::Vector2f(minX, minY), sf::Vector2f(maxX - minX, maxY - minY));

        int code1 = computeOutCode(x1, y1, tempRect);
        int code2 = computeOutCode(x2, y2, tempRect);

        while (true) {
            if (!(code1 | code2)) {
                return std::make_pair(std::make_pair(x1, y1), std::make_pair(x2, y2));
            }
            if (code1 & code2) {
                return std::nullopt;
            }

            int codeOut = code1 ? code1 : code2;
            float x = 0, y = 0;

            // ProtecciÃ³n contra divisiÃ³n por (casi) cero
            if (codeOut & TOP) {
                if (std::abs(y2 - y1) > 1e-5f) x = x1 + (x2 - x1) * (minY - y1) / (y2 - y1);
                else x = x1;
                y = minY;
            }
            else if (codeOut & BOTTOM) {
                if (std::abs(y2 - y1) > 1e-5f) x = x1 + (x2 - x1) * (maxY - y1) / (y2 - y1);
                else x = x1;
                y = maxY;
            }
            else if (codeOut & LEFT) {
                if (std::abs(x2 - x1) > 1e-5f) y = y1 + (y2 - y1) * (minX - x1) / (x2 - x1);
                else y = y1;
                x = minX;
            }
            else { // RIGHT
                if (std::abs(x2 - x1) > 1e-5f) y = y1 + (y2 - y1) * (maxX - x1) / (x2 - x1);
                else y = y1;
                x = maxX;
            }

            if (codeOut == code1) {
                x1 = x; y1 = y;
                code1 = computeOutCode(x1, y1, tempRect);
                intersections.emplace_back(x, y);
            }
            else {
                x2 = x; y2 = y;
                code2 = computeOutCode(x2, y2, tempRect);
                intersections.emplace_back(x, y);
            }
        }
    }

    static void kochRecursive(Graph& g, int depth,
        float x1, float y1,
        float x2, float y2) {
        if (depth == 0) {
            uint64_t id1 = g.addNode(x1, y1);
            uint64_t id2 = g.addNode(x2, y2);
            float dist = std::hypot(x2 - x1, y2 - y1);
            g.addEdge(id1, id2, dist);
            return;
        }

        // Dividir el segmento en tres partes
        float dx = (x2 - x1) / 3.0f;
        float dy = (y2 - y1) / 3.0f;

        float p1x = x1 + dx;
        float p1y = y1 + dy;
        float p3x = x1 + 2.0f * dx;
        float p3y = y1 + 2.0f * dy;

        // Punto medio elevado para el triÃ¡ngulo equilÃ¡tero
        float angle = -60.0f * 3.14159265f / 180.0f; // -60 grados
        float cosA = std::cos(angle);
        float sinA = std::sin(angle);
        float p2x = p1x + (p3x - p1x) * cosA - (p3y - p1y) * sinA;
        float p2y = p1y + (p3x - p1x) * sinA + (p3y - p1y) * cosA;

        // RecursiÃ³n en los 4 segmentos
        kochRecursive(g, depth - 1, x1, y1, p1x, p1y);
        kochRecursive(g, depth - 1, p1x, p1y, p2x, p2y);
        kochRecursive(g, depth - 1, p2x, p2y, p3x, p3y);
        kochRecursive(g, depth - 1, p3x, p3y, x2, y2);
    }

    void koch(Graph& g, int depth, float x1, float y1, float x2, float y2) {
        kochRecursive(g, depth, x1, y1, x2, y2);
    }

    void kochTriangle(Graph& g, int depth,
        float x1, float y1,
        float x2, float y2,
        float x3, float y3) {
        koch(g, depth, x1, y1, x2, y2);
        koch(g, depth, x2, y2, x3, y3);
        koch(g, depth, x3, y3, x1, y1);

    struct Triangle { float ax, ay, bx, by, cx, cy; };

    static Triangle reflectTriangle(const Triangle& tri, int side) {
        float ox, oy, v1x, v1y, v2x, v2y;
        if (side == 0) {
            v1x = tri.ax; v1y = tri.ay; v2x = tri.bx; v2y = tri.by; ox = tri.cx;  oy = tri.cy;
        }
        else if (side == 1) {
            v1x = tri.bx; v1y = tri.by; v2x = tri.cx; v2y = tri.cy; ox = tri.ax;  oy = tri.ay;
        }
        else {
            v1x = tri.cx; v1y = tri.cy; v2x = tri.ax; v2y = tri.ay; ox = tri.bx;  oy = tri.by;
        }
        float nox = v1x + v2x - ox;
        float noy = v1y + v2y - oy;
        if (side == 0)      return { nox, noy, tri.bx, tri.by, tri.ax, tri.ay };
        else if (side == 1) return { tri.ax, tri.ay, nox, noy, tri.cx, tri.cy };
        else                return { tri.ax, tri.ay, tri.bx, tri.by, nox, noy };
    }

    static bool triangleIntersectsBounds(const Triangle& tri, const sf::FloatRect& bounds) {
        auto inside = [&](float x, float y) {
            return x >= bounds.position.x && x <= bounds.position.x + bounds.size.x &&
                y >= bounds.position.y && y <= bounds.position.y + bounds.size.y;
            };
        return inside(tri.ax, tri.ay) || inside(tri.bx, tri.by) || inside(tri.cx, tri.cy);
    }

    static std::vector<uint64_t> createAnchors(Graph& g, const sf::FloatRect& bounds) {
        std::vector<uint64_t> anchors;
        float minX = bounds.position.x;
        float maxX = bounds.position.x + bounds.size.x;
        float minY = bounds.position.y;
        float maxY = bounds.position.y + bounds.size.y;
        float spacing = ANCHOR_SPACING;

        for (float x = minX; x <= maxX; x += spacing) anchors.push_back(g.addNode(x, minY));
        for (float y = minY + spacing; y <= maxY; y += spacing) anchors.push_back(g.addNode(maxX, y));
        for (float x = maxX - spacing; x >= minX; x -= spacing) anchors.push_back(g.addNode(x, maxY));
        for (float y = maxY - spacing; y >= minY; y -= spacing) anchors.push_back(g.addNode(minX, y));
        return anchors;
    }

    static void connectAnchors(Graph& g, const std::vector<uint64_t>& anchors) {
        if (anchors.size() < 2) return;
        for (size_t i = 0; i < anchors.size(); ++i) {
            uint64_t a = anchors[i];
            uint64_t b = anchors[(i + 1) % anchors.size()];
            auto itA = g.getNodes().find(a);
            auto itB = g.getNodes().find(b);
            if (itA != g.getNodes().end() && itB != g.getNodes().end()) {
                float dx = itA->second.getX() - itB->second.getX();
                float dy = itA->second.getY() - itB->second.getY();
                g.addEdge(a, b, std::hypot(dx, dy));
            }
        }
    }

    static void clipAndAddTriangle(Graph& g, const sf::FloatRect& bounds,
        float ax, float ay, float bx, float by, float cx, float cy, int localDepth) {
        if (localDepth > 0) {
            entities::algorithms::fractal::sierpinski(g, localDepth, ax, ay, bx, by, cx, cy, 1);
        }

    static void attachToAnchors(Graph& g, const std::vector<uint64_t>& anchors, const sf::FloatRect& bounds) {
        const auto& nodes = g.getNodes();
        std::unordered_set<uint64_t> anchorSet(anchors.begin(), anchors.end());

        for (uint64_t anchorId : anchors) {
            auto itAnchor = nodes.find(anchorId);
            if (itAnchor == nodes.end()) continue;

            float ax = itAnchor->second.getX();
            float ay = itAnchor->second.getY();
            uint64_t bestNode = 0;
            float minDistSq = std::numeric_limits<float>::max();

            for (const auto& [nodeId, node] : nodes) {
                if (anchorSet.count(nodeId)) continue;
                float nx = node.getX(), ny = node.getY();
                if (nx < bounds.position.x || nx > bounds.position.x + bounds.size.x ||
                    ny < bounds.position.y || ny > bounds.position.y + bounds.size.y) continue;

                float distSq = (nx - ax) * (nx - ax) + (ny - ay) * (ny - ay);
                if (distSq < minDistSq) {
                    minDistSq = distSq;
                    bestNode = nodeId;
                }
            }
            if (bestNode != 0 && minDistSq < (250.0f * 250.0f)) {
                g.addEdge(anchorId, bestNode, std::sqrt(minDistSq));
            }
        }
    }

    static void fillPlane(Graph& g, const std::vector<uint64_t>& anchors, const sf::FloatRect& bounds,
        const Triangle& seed, int maxDepth, std::function<int(float, float)> depthFunction) {
        std::deque<Triangle> queue;
        std::unordered_set<uint64_t> visitedCenters;
        const size_t MAX_TRIANGLES = 15000;

        auto centerID = [](const Triangle& t) -> uint64_t {
            float cx = (t.ax + t.bx + t.cx) / 3.0f;
            float cy = (t.ay + t.by + t.cy) / 3.0f;
            int32_t qx = static_cast<int32_t>(std::round(cx * 1.0f));
            int32_t qy = static_cast<int32_t>(std::round(cy * 1.0f));
            auto zigzag = [](int32_t n) -> uint32_t { return (n << 1) ^ (n >> 31); };
            return (static_cast<uint64_t>(zigzag(qx)) << 32) | zigzag(qy);
            };

        uint64_t seedID = centerID(seed);
        visitedCenters.insert(seedID);
        queue.push_back(seed);

        while (!queue.empty() && visitedCenters.size() < MAX_TRIANGLES) {
            Triangle tri = queue.front();
            queue.pop_front();
            float cx = (tri.ax + tri.bx + tri.cx) / 3.0f;
            float cy = (tri.ay + tri.by + tri.cy) / 3.0f;
            int localDepth = depthFunction(cx, cy);

            clipAndAddTriangle(g, bounds, tri.ax, tri.ay, tri.bx, tri.by, tri.cx, tri.cy, localDepth);

            for (int side = 0; side < 3; ++side) {
                Triangle neighbor = reflectTriangle(tri, side);
                if (!triangleIntersectsBounds(neighbor, bounds)) continue;
                uint64_t nID = centerID(neighbor);
                if (visitedCenters.find(nID) == visitedCenters.end()) {
                    visitedCenters.insert(nID);
                    queue.push_back(neighbor);
                }
            }
        }
    }

    static void drawStringArtRect(Graph& g, float rx, float ry, float rw, float rh, int N) {
        for (int dir = 0; dir < 4; ++dir) {
            for (int i = 0; i <= N; ++i) {
                float t = static_cast<float>(i) / N;
                float invT = 1.0f - t;
                float x1, y1, x2, y2;
                if (dir == 0) { x1 = rx + t * rw; y1 = ry; x2 = rx; y2 = ry + invT * rh; }
                else if (dir == 1) { x1 = rx + rw - t * rw; y1 = ry; x2 = rx + rw; y2 = ry + invT * rh; }
                else if (dir == 2) { x1 = rx + rw - t * rw; y1 = ry + rh; x2 = rx + rw; y2 = ry + rh - invT * rh; }
                else { x1 = rx + t * rw; y1 = ry + rh; x2 = rx; y2 = ry + rh - invT * rh; }

                uint64_t id1 = g.addNode(x1, y1);
                uint64_t id2 = g.addNode(x2, y2);
                if (id1 != id2) g.addEdge(id1, id2, std::hypot(x2 - x1, y2 - y1));
            }
        }
    }

    void mandelbrotLoomPattern(Graph& g, int depth) {
        const float minX = -LOOM_WIDTH / 2 + MARGIN;
        const float maxX = LOOM_WIDTH / 2 - MARGIN;
        const float minY = -LOOM_HEIGHT / 2 + MARGIN;
        const float maxY = LOOM_HEIGHT / 2 - MARGIN;
        sf::FloatRect bounds(sf::Vector2f(minX, minY), sf::Vector2f(maxX - minX, maxY - minY));

        auto anchors = createAnchors(g, bounds);
        connectAnchors(g, anchors);

        const int gridW = 80, gridH = 60, maxIter = 64;
        auto pts = evalMandelbrot(bounds, gridW, gridH, maxIter);

        // Crear nodos solo en puntos interesantes
        std::vector<uint64_t> nodeId(gridW * gridH, UINT64_MAX);
        for (int i = 0; i < static_cast<int>(pts.size()); ++i) {
            const auto& p = pts[i];
            bool isEscape = (p.escape != -1);
            bool interesting = isEscape ? (p.escape >= 2 && p.escape <= maxIter - 2) : true;
            if (interesting) {
                nodeId[i] = g.addNode(static_cast<float>(p.px), static_cast<float>(p.py));
            }
        }

        // Conectar vecinos (4 direcciones)
        const int dx[] = { 1, 0, 1, 1 };
        const int dy[] = { 0, 1, 1, -1 };
        for (int gy = 0; gy < gridH; ++gy) {
            for (int gx = 0; gx < gridW; ++gx) {
                int idx = gy * gridW + gx;
                if (nodeId[idx] == UINT64_MAX) continue;
                for (int d = 0; d < 4; ++d) {
                    int nx = gx + dx[d], ny = gy + dy[d];
                    if (nx < 0 || nx >= gridW || ny < 0 || ny >= gridH) continue;
                    int nidx = ny * gridW + nx;
                    if (nodeId[nidx] == UINT64_MAX) continue;
                    float w = std::hypot(pts[idx].px - pts[nidx].px, pts[idx].py - pts[nidx].py);
                    g.addEdge(nodeId[idx], nodeId[nidx], w);
                }
            }
        }

        attachToAnchors(g, anchors, bounds);
    }

    // Recolecta todas las aristas del grafo con sus pesos
    struct TempEdge { uint64_t from, to; float weight; };
    static std::vector<TempEdge> collectEdges(const Graph& g) {
        std::vector<TempEdge> edges;
        for (const auto& [idA, nodeA] : g.getNodes()) {
            for (uint64_t idB : nodeA.getNeighbors()) {
                if (idA < idB) { // evitar duplicados
                    edges.push_back({ idA, idB, nodeA.getWeightTo(idB) });
                }
            }
        }
        return edges;
    }

    // Union-Find local
    static uint64_t ufFind(std::unordered_map<uint64_t, uint64_t>& parent, uint64_t x) {
        if (parent[x] != x) parent[x] = ufFind(parent, parent[x]);
        return parent[x];
    }
    static bool ufUnion(std::unordered_map<uint64_t, uint64_t>& parent, std::unordered_map<uint64_t, int>& rank, uint64_t a, uint64_t b) {
        uint64_t ra = ufFind(parent, a), rb = ufFind(parent, b);
        if (ra == rb) return false;
        if (rank[ra] < rank[rb]) std::swap(ra, rb);
        parent[rb] = ra;
        if (rank[ra] == rank[rb]) ++rank[ra];
        return true;
    }

    void kruskalLoomPattern(Graph& g, int depth) {
        const float minX = -LOOM_WIDTH / 2 + MARGIN;
        const float maxX = LOOM_WIDTH / 2 - MARGIN;
        const float minY = -LOOM_HEIGHT / 2 + MARGIN;
        const float maxY = LOOM_HEIGHT / 2 - MARGIN;
        sf::FloatRect bounds(sf::Vector2f(minX, minY), sf::Vector2f(maxX - minX, maxY - minY));

        auto anchors = createAnchors(g, bounds);
        connectAnchors(g, anchors);

        // Generar puntos aleatorios (semilla controlada por SeedManager)
        auto& rng = SeedManager::instance().rng();
        std::uniform_real_distribution<float> xDist(minX + 10, maxX - 10);
        std::uniform_real_distribution<float> yDist(minY + 10, maxY - 10);
        int numPoints = 200;
        std::vector<uint64_t> ids;
        for (int i = 0; i < numPoints; ++i) {
            float x = xDist(rng), y = yDist(rng);
            ids.push_back(g.addNode(x, y));
        }

        // Grafo completo basado en distancia: conectar puntos dentro de un radio
        // (esto es un proxy rÃ¡pido en lugar de Delaunay)
        float radius = 250.0f;
        for (size_t i = 0; i < ids.size(); ++i) {
            auto itI = g.getNodes().find(ids[i]);
            if (itI == g.getNodes().end()) continue;
            float xi = itI->second.getX(), yi = itI->second.getY();
            for (size_t j = i + 1; j < ids.size(); ++j) {
                auto itJ = g.getNodes().find(ids[j]);
                if (itJ == g.getNodes().end()) continue;
                float xj = itJ->second.getX(), yj = itJ->second.getY();
                float d = std::hypot(xi - xj, yi - yj);
                if (d < radius) {
                    g.addEdge(ids[i], ids[j], d);
                }
            }
        }

        // Recolectar aristas y ejecutar Kruskal con prioridad a los anclajes
        auto edges = collectEdges(g);
        // Marcar aristas que tocan anclas
        std::unordered_set<uint64_t> anchorSet(anchors.begin(), anchors.end());
        std::stable_sort(edges.begin(), edges.end(),
            [&](const TempEdge& a, const TempEdge& b) {
                bool aAnch = anchorSet.count(a.from) || anchorSet.count(a.to);
                bool bAnch = anchorSet.count(b.from) || anchorSet.count(b.to);
                if (aAnch != bAnch) return aAnch > bAnch;
                return a.weight < b.weight;
            });

        std::unordered_map<uint64_t, uint64_t> parent;
        std::unordered_map<uint64_t, int> rank;
        for (auto id : ids) parent[id] = id;

        // Limpiar grafo y aÃ±adir solo aristas del MST
        // (no podemos borrar nodos, pero sÃ­ podemos eliminar aristas sobrantes)
        // Para simplificar, construimos un nuevo grafo temporal y luego movemos los datos.
        // En su lugar, marcaremos las aristas a conservar y eliminaremos las demÃ¡s.
        std::unordered_set<std::pair<uint64_t, uint64_t>, PairHash> mstEdges;
        for (const auto& e : edges) {
            if (ufUnion(parent, rank, e.from, e.to)) {
                mstEdges.insert({ std::min(e.from, e.to), std::max(e.from, e.to) });
            }
        }

        // Eliminar aristas que no estÃ¡n en el MST
        for (const auto& [idA, nodeA] : g.getNodes()) {
            for (uint64_t idB : nodeA.getNeighbors()) {
                if (idA < idB) {
                    auto key = std::make_pair(idA, idB);
                    if (mstEdges.find(key) == mstEdges.end()) {
                        g.deleteEdge(idA, idB);
                    }
                }
            }
        }

        attachToAnchors(g, anchors, bounds);
    }

    void randomSierpinski(Graph& g, int& depth, int& outerLayers, bool& vertexMode) {
        thread_local std::mt19937 rng(std::random_device{}());

        std::uniform_int_distribution<int> depthDist(2, 4);
        depth = depthDist(rng);

        std::uniform_int_distribution<int> modeDist(0, 1);
        vertexMode = (modeDist(rng) == 1);

        constexpr float canvasW = 1200.0f;
        constexpr float canvasH = 900.0f;

        constexpr float cellW = 300.0f;
        constexpr float cellH = 300.0f;

        int cols = static_cast<int>(canvasW / cellW) + 1;
        int rows = static_cast<int>(canvasH / cellH) + 1;

        std::uniform_real_distribution<float> jitter(-60.0f, 60.0f);
        std::uniform_real_distribution<float> radiusDist(30.0f, 120.0f);
        std::uniform_real_distribution<float> angleDist(0.0f, 6.28318f);

        std::uniform_int_distribution<int> outerDist(4, 8);

        int triangleCount = 0;
        for (int row = 0; row < rows; ++row) {
            for (int col = 0; col < cols; ++col) {
                float centerX = -canvasW / 2 + col * cellW + cellW / 2;
                float centerY = -canvasH / 2 + row * cellH + cellH / 2;

                float ox = jitter(rng);
                float oy = jitter(rng);
                float cx = centerX + ox;
                float cy = centerY + oy;

                float x1, y1, x2, y2, x3, y3;
                do {
                    auto randPt = [&](float baseX, float baseY) {
                        float a = angleDist(rng);
                        float r = radiusDist(rng);
                        return std::make_pair(baseX + r * cosf(a), baseY + r * sinf(a));
                        };
                    auto [px1, py1] = randPt(cx, cy);
                    auto [px2, py2] = randPt(cx, cy);
                    auto [px3, py3] = randPt(cx, cy);
                    x1 = px1; y1 = py1;
                    x2 = px2; y2 = py2;
                    x3 = px3; y3 = py3;
                    float area = 0.5f * std::abs(x1 * (y2 - y3) + x2 * (y3 - y1) + x3 * (y1 - y2));
                    if (area > 500.0f) break;
                } while (true);

                if (vertexMode) {
                    expandAtVertices(g, depth, x1, y1, x2, y2, x3, y3);
                }
                else {
                    int layers = outerDist(rng);
                    expandSierpinski(g, depth, layers, x1, y1, x2, y2, x3, y3);
                }
                ++triangleCount;
            }
        }

        outerLayers = triangleCount;
    }

    // ------------------------------------------------------------
    // Curva de Koch DOBLE (expansiÃ³n en rombo)
    // ------------------------------------------------------------
    static void kochEdge(Graph& g, int depth, float x1, float y1, float x2, float y2) {
        if (depth == 0) {
            uint64_t id1 = g.addNode(x1, y1);
            uint64_t id2 = g.addNode(x2, y2);
            g.addEdge(id1, id2, std::hypot(x2 - x1, y2 - y1));
            return;
        }

        float dx = x2 - x1;
        float dy = y2 - y1;

        // Puntos a 1/3 y 2/3 del segmento
        float p1x = x1 + dx / 3.0f;
        float p1y = y1 + dy / 3.0f;
        float p3x = x1 + 2.0f * dx / 3.0f;
        float p3y = y1 + 2.0f * dy / 3.0f;

        // Vector del tercio central (de P1 a P3)
        float vx = p3x - p1x;   // = dx / 3
        float vy = p3y - p1y;   // = dy / 3

        // RotaciÃ³n +60Â° (pico exterior)
        float out_px = p1x + vx * 0.5f - vy * 0.866025f;
        float out_py = p1y + vx * 0.866025f + vy * 0.5f;

        // RotaciÃ³n -60Â° (pico interior)
        float in_px = p1x + vx * 0.5f + vy * 0.866025f;
        float in_py = p1y - vx * 0.866025f + vy * 0.5f;

        // 6 llamadas recursivas (diamante)
        kochEdge(g, depth - 1, x1, y1, p1x, p1y);        // A â†’ P1
        kochEdge(g, depth - 1, p1x, p1y, out_px, out_py); // P1 â†’ Pico externo
        kochEdge(g, depth - 1, out_px, out_py, p3x, p3y); // Pico externo â†’ P3
        kochEdge(g, depth - 1, p1x, p1y, in_px, in_py);   // P1 â†’ Pico interno
        kochEdge(g, depth - 1, in_px, in_py, p3x, p3y);   // Pico interno â†’ P3
        kochEdge(g, depth - 1, p3x, p3y, x2, y2);         // P3 â†’ B
    }

    // ------------------------------------------------------------
    // Sierpinski original con bordes Koch-doble
    // ------------------------------------------------------------
    void sierpinski(Graph& g, int depth, float x1, float y1, float x2, float y2, float x3, float y3, int kochDepth) {
        if (depth == 0) {
            kochEdge(g, kochDepth, x1, y1, x2, y2);
            kochEdge(g, kochDepth, x2, y2, x3, y3);
            kochEdge(g, kochDepth, x3, y3, x1, y1);
        }
        else {
            float m12_x = (x1 + x2) / 2.0f;
            float m12_y = (y1 + y2) / 2.0f;
            float m23_x = (x2 + x3) / 2.0f;
            float m23_y = (y2 + y3) / 2.0f;
            float m31_x = (x3 + x1) / 2.0f;
            float m31_y = (y1 + y3) / 2.0f;

            sierpinski(g, depth - 1, x1, y1, m12_x, m12_y, m31_x, m31_y, kochDepth);
            sierpinski(g, depth - 1, m12_x, m12_y, x2, y2, m23_x, m23_y, kochDepth);
            sierpinski(g, depth - 1, m31_x, m31_y, m23_x, m23_y, x3, y3, kochDepth);
        }
    }

    sf::Vector2f SfmlRenderer::worldToScreen(const sf::FloatRect& bounds,
        const sf::Vector2u& winSize,
        float wx, float wy) {
        float scaleX = winSize.x / bounds.size.x;
        float scaleY = winSize.y / bounds.size.y;
        float x = (wx - bounds.position.x) * scaleX;
        float y = (bounds.position.y + bounds.size.y - wy) * scaleY;
        return { x, y };
    }

    void SfmlRenderer::drawAnimated(sf::RenderTarget& target, const Graph& graph,
        float progress,
        sf::Color revealedNodeColor, sf::Color normalNodeColor,
        sf::Color edgeColor) {
        const auto& nodesMap = graph.getNodes();
        if (nodesMap.empty()) return;

        sf::FloatRect bounds(sf::Vector2f(-600.0f, -450.0f), sf::Vector2f(1200.0f, 900.0f));
        sf::Vector2u winSize = target.getSize();

        auto getPos = [&](uint64_t id) -> sf::Vector2f {
            auto it = nodesMap.find(id);
            if (it != nodesMap.end())
                return worldToScreen(bounds, winSize, it->second.getX(), it->second.getY());
            return { 0, 0 };
            };

        // ---- Aristas con VertexArray Ãºnico ----
        size_t maxEdges = graph.getEdgeCreationOrder().size();
        sf::VertexArray edgeArray(sf::PrimitiveType::Lines, maxEdges * 2);
        size_t edgeIdx = 0;
        for (const auto& [idA, nodeA] : nodesMap) {
            sf::Vector2f posA = getPos(idA);
            for (uint64_t idB : nodeA.getNeighbors()) {
                if (idA > idB) continue;
                auto itB = nodesMap.find(idB);
                if (itB == nodesMap.end()) continue;
                sf::Vector2f posB = getPos(idB);
                if (edgeIdx + 1 < edgeArray.getVertexCount()) {
                    edgeArray[edgeIdx].position = posA;
                    edgeArray[edgeIdx].color = edgeColor;
                    edgeArray[edgeIdx + 1].position = posB;
                    edgeArray[edgeIdx + 1].color = edgeColor;
                    edgeIdx += 2;
                }
                else {
                    edgeArray.append(sf::Vertex(posA, edgeColor));
                    edgeArray.append(sf::Vertex(posB, edgeColor));
                }
            }
        }
        target.draw(edgeArray);

        // Dibujar marco negro ANTES de los nodos
        drawFrame(target, winSize);

        // ---- Nodos revelados ----
        const auto& creationOrder = graph.getNodeCreationOrder();
        size_t totalNodes = creationOrder.size();
        size_t revealedCount = static_cast<size_t>(progress * totalNodes);
        std::unordered_set<uint64_t> revealedSet;
        for (size_t i = 0; i < revealedCount && i < totalNodes; ++i)
            revealedSet.insert(creationOrder[i]);

        // ---- Nodos con VertexArray de Puntos ----
        sf::VertexArray nodeArray(sf::PrimitiveType::Points, nodesMap.size());
        size_t nodeIdx = 0;
        for (const auto& [id, node] : nodesMap) {
            sf::Vector2f pos = getPos(id);
            nodeArray[nodeIdx].position = pos;
            nodeArray[nodeIdx].color = revealedSet.count(id) ? revealedNodeColor : normalNodeColor;
            ++nodeIdx;
        }
        sf::RenderStates states;
        states.shader = &getPointShader();
        target.draw(nodeArray, states);
    }

    void SfmlRenderer::drawBuildAnimation(sf::RenderTarget& target, const Graph& graph,
        float progress,
        sf::Color baseNodeColor, sf::Color brightNodeColor,
        sf::Color baseEdgeColor, sf::Color brightEdgeColor) {
        const auto& nodesMap = graph.getNodes();
        if (nodesMap.empty()) return;

        sf::FloatRect bounds(sf::Vector2f(-600.0f, -450.0f), sf::Vector2f(1200.0f, 900.0f));
        sf::Vector2u winSize = target.getSize();

        auto getPos = [&](uint64_t id) -> sf::Vector2f {
            auto it = nodesMap.find(id);
            if (it != nodesMap.end())
                return worldToScreen(bounds, winSize, it->second.getX(), it->second.getY());
            return { 0, 0 };
            };

        const auto& nodeOrder = graph.getNodeCreationOrder();
        const auto& edgeOrder = graph.getEdgeCreationOrder();
        size_t totalNodes = nodeOrder.size();
        size_t totalEdges = edgeOrder.size();

        size_t nodeCut = static_cast<size_t>(progress * totalNodes);
        size_t edgeCut = static_cast<size_t>(progress * totalEdges);

        // ---- Aristas con brillo progresivo ----
        sf::VertexArray edgeArray(sf::PrimitiveType::Lines, edgeCut * 2);
        size_t edgeIdx = 0;
        for (size_t i = 0; i < edgeCut; ++i) {
            uint64_t idA = edgeOrder[i].first;
            uint64_t idB = edgeOrder[i].second;
            auto itA = nodesMap.find(idA);
            auto itB = nodesMap.find(idB);
            if (itA == nodesMap.end() || itB == nodesMap.end()) continue;

            sf::Vector2f posA = getPos(idA);
            sf::Vector2f posB = getPos(idB);

            float t = static_cast<float>(i) / totalEdges;
            sf::Color edgeCol(
                static_cast<uint8_t>(baseEdgeColor.r + (brightEdgeColor.r - baseEdgeColor.r) * t),
                static_cast<uint8_t>(baseEdgeColor.g + (brightEdgeColor.g - baseEdgeColor.g) * t),
                static_cast<uint8_t>(baseEdgeColor.b + (brightEdgeColor.b - baseEdgeColor.b) * t)
            );

            if (edgeIdx + 1 < edgeArray.getVertexCount()) {
                edgeArray[edgeIdx].position = posA;
                edgeArray[edgeIdx].color = edgeCol;
                edgeArray[edgeIdx + 1].position = posB;
                edgeArray[edgeIdx + 1].color = edgeCol;
                edgeIdx += 2;
            }
            else {
                edgeArray.append(sf::Vertex(posA, edgeCol));
                edgeArray.append(sf::Vertex(posB, edgeCol));
            }
        }
        target.draw(edgeArray);

        // Dibujar marco negro ANTES de los nodos
        drawFrame(target, winSize);

        // ---- Nodos con brillo progresivo ----
        sf::VertexArray nodeArray(sf::PrimitiveType::Points, nodeCut);
        size_t nodeIdx = 0;
        for (size_t i = 0; i < nodeCut; ++i) {
            uint64_t id = nodeOrder[i];
            sf::Vector2f pos = getPos(id);
            float t = static_cast<float>(i) / totalNodes;
            sf::Color nodeCol(
                static_cast<uint8_t>(baseNodeColor.r + (brightNodeColor.r - baseNodeColor.r) * t),
                static_cast<uint8_t>(baseNodeColor.g + (brightNodeColor.g - baseNodeColor.g) * t),
                static_cast<uint8_t>(baseNodeColor.b + (brightNodeColor.b - baseNodeColor.b) * t)
            );
            nodeArray[nodeIdx].position = pos;
            nodeArray[nodeIdx].color = nodeCol;
            ++nodeIdx;
        }

        sf::RenderStates states;
        states.shader = &getPointShader();
        target.draw(nodeArray, states);
    }
